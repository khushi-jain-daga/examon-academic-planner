function moduleWindow(plan,moduleId){const a=(plan.sessions||[]).filter(s=>s.moduleId===moduleId).map(s=>s.date).sort();return {start:a[0]||'',end:a[a.length-1]||''}}
function importRollingPlan(source,newStart){
  const d=blankDraft();
  d.examName=source.examName; d.branch=source.branch; d.planType=source.planType; d.contentFlags=[...(source.contentFlags||[])];
  d.startDate=newStart||iso(new Date()); d.selectedFacultyIds=[...new Set(source.modules.flatMap(m=>m.facultyIds||[]))];
  const grouped={};
  source.modules.forEach(m=>(grouped[m.track||'Track A']??=[]).push(m));
  d.modules=[];
  Object.entries(grouped).forEach(([track,mods])=>{
    const live=[],completed=[];
    mods.forEach(m=>{
      const w=moduleWindow(source,m.id);
      const c={...deep(m),id:uid('mod'),customStart:'',startMode:'auto',weekendOverride:false};
      if(w.end && w.end<d.startDate) completed.push(c); else live.push(c);
    });
    const ordered=[...live,...completed];
    ordered.forEach((m,i)=>{m.priority=i+1;m.track=track;d.modules.push(m)});
  });
  return d;
}
