function toDate(s){const [y,m,d]=s.split('-').map(Number);return new Date(y,m-1,d)}
function iso(d){return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`}
function addDays(d,n){const x=new Date(d);x.setDate(x.getDate()+n);return x}
function dayName(s){return DAY[toDate(s).getDay()]}
function pretty(s){if(!s)return '—';return toDate(s).toLocaleDateString('en-GB',{day:'2-digit',month:'short',year:'numeric'})}
function validDates(start,count,workingDays,blocked=[],weekend=false){let d=toDate(start),out=[],guard=0;const allow=new Set(weekend?[...new Set([...workingDays,0,6])]:workingDays),blk=new Set(blocked);while(out.length<count&&guard<5000){if(allow.has(d.getDay())&&!blk.has(iso(d)))out.push(iso(d));d=addDays(d,1);guard++}return out}
function nextWorking(start,days,blocked){return validDates(iso(addDays(toDate(start),1)),1,days,blocked,false)[0]}
function generatePlanSessions(plan){
  const errors=[]; for(const m of plan.modules){if(!m.startTime)errors.push(`${m.subjectName}: time required`);if(!m.facultyIds.length)errors.push(`${m.subjectName}: faculty required`);if(!m.classes||m.classes<1)errors.push(`${m.subjectName}: classes required`)} if(errors.length)return {errors,sessions:[]};
  const tracks={};plan.modules.forEach(m=>(tracks[m.track]??=[]).push(m));Object.values(tracks).forEach(a=>a.sort((x,y)=>x.priority-y.priority));let sessions=[];
  for(const mods of Object.values(tracks)){let cursor=plan.startDate;for(const m of mods){const start=m.startMode==='custom'&&m.customStart?m.customStart:cursor;const dates=validDates(start,m.classes,m.workingDays?.length?m.workingDays:plan.workingDays,state.settings.holidays,m.weekendOverride);dates.forEach((date,i)=>sessions.push({id:uid('ses'),moduleId:m.id,classIndex:i+1,date,startTime:m.startTime,duration:m.duration,facultyIds:[...m.facultyIds],subjectName:m.subjectName,category:m.category,topic:`${m.subjectName} - ${i+1}`,locked:false,exception:[0,6].includes(toDate(date).getDay())?'Weekend':''}));if(dates.length)cursor=nextWorking(dates.at(-1),m.workingDays?.length?m.workingDays:plan.workingDays,state.settings.holidays)}}
  return {errors:[],sessions:sessions.sort((a,b)=>a.date.localeCompare(b.date)||a.startTime.localeCompare(b.startTime))}
}

function overlaps(a,b){return a.date===b.date&&tmin(a.startTime)<tmin(b.startTime)+b.duration&&tmin(b.startTime)<tmin(a.startTime)+a.duration}
function conflicts(){let all=[];state.plans.filter(p=>p.status!=='Archived').forEach(p=>p.sessions.forEach(s=>all.push({p,s})));let out=[];for(let i=0;i<all.length;i++)for(let j=i+1;j<all.length;j++){const a=all[i],b=all[j];if(!overlaps(a.s,b.s))continue;const shared=a.s.facultyIds.filter(x=>b.s.facultyIds.includes(x));if(shared.length)out.push({type:'Faculty',date:a.s.date,msg:`${facultyNames(shared)} double-booked: ${a.s.subjectName} (${a.p.batchName}) vs ${b.s.subjectName} (${b.p.batchName})`});else if(a.p.id===b.p.id)out.push({type:'Batch',date:a.s.date,msg:`Batch overlap: ${a.s.subjectName} vs ${b.s.subjectName}`})}return out}
